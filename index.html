async function finalizarCompra() { if (carrinho.length === 0) { alert('Adicione itens ao carrinho primeiro!'); return; } try { // 1. Registra a venda no banco para dar baixa no estoque const pedido = { origem: 'E-COMMERCE', metodo_pagamento: 'MERCADO_PAGO', itens: carrinho }; const resVenda = await fetch(`${API_URL}/vendas`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(pedido) }); const dataVenda = await resVenda.json(); if (!resVenda.ok) { alert(`Erro no pedido: ${dataVenda.error}`); return; } // 2. Gera o checkout oficial do Mercado Pago (PIX + Cartão) const resCheckout = await fetch(`${API_URL}/pagamento/checkout`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ itens: carrinho, emailCliente: 'cliente@fragranciasfabi.com.br' }) }); const dataCheckout = await resCheckout.json(); if (dataCheckout.init_point) { carrinho = []; atualizarCarrinho(); fecharCarrinho(); // Redireciona para o pagamento no Mercado Pago window.location.href = dataCheckout.init_point; } else { alert('Erro ao gerar tela de pagamento.'); } } catch (err) { console.error(err); alert('Erro ao conectar com o servidor.'); } }
